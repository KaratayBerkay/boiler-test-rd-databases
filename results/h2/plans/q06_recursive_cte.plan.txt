WITH  RECURSIVE "tree"("id", "parent_id", "name", "depth", "path") AS (
    (SELECT
        "id",
        "parent_id",
        "name",
        0,
        CAST("name" AS CHARACTER VARYING(400))
    FROM "public"."categories"
    WHERE "parent_id" IS NULL)
    UNION ALL
    (SELECT
        "c"."id",
        "c"."parent_id",
        "c"."name",
        "t"."depth" + 1,
        CAST("t"."path" || ' > ' || "c"."name" AS CHARACTER VARYING(400))
    FROM "public"."categories" "c"
    INNER JOIN "tree" "t"
        ON 1=1
    WHERE "c"."parent_id" = "t"."id")
)
SELECT
    "id",
    "depth",
    "path"
FROM "tree" "tree"
    /* null */
ORDER BY 3
